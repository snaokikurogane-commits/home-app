function drawChart(items) { const ctx = document.getElementById('weightChart').getContext('2d'); if(myChart) myChart.destroy(); myChart = new Chart(ctx, { type: 'line', data: { labels: items.map(i => i.date), datasets: [ { label: '尚輝', data: items.map(i => i.naoki), borderColor: '#5ab48e', backgroundColor: '#5ab48e', tension: 0.3 }, { label: '尚輝目標', data: items.map(i => i.naokiGoal), borderColor: '#5ab48e', borderDash: [5, 5], pointRadius: 0, fill: false }, { label: '真輝', data: items.map(i => i.maki), borderColor: '#F2C94C', backgroundColor: '#F2C94C', tension: 0.3 }, { label: '真輝目標', data: items.map(i => i.makiGoal), borderColor: '#F2C94C', borderDash: [5, 5], pointRadius: 0, fill: false } ] }, options: { responsive: true, maintainAspectRatio: false, plugins: { legend: { position: 'bottom' } }, // ★ ここに「40kg〜80kgで固定」の設定を追加しました！ scales: { y: { min: 40, max: 80 } } } }); }
